function UpdateAlpha.hierarchical(hdpObj):
    // hdpObj has a list of individual DPs (indDP_list) 
    // and a global concentration parameter gamma.

    // Update alpha for each individual DP
    for each indDP in hdpObj.indDP_list:
        indDP = UpdateAlpha.default(indDP) // or its specific type if dispatch needed
        // Ensure consistency if alpha is also stored in mixingDistribution for indDP
        indDP.mixing_distribution.alpha = indDP.alpha 
    
    // Update global concentration parameter gamma
    // This uses the same logic as UpdateAlpha.default (update_concentration)
    // but with inputs relevant to the global level of the HDP.
    
    // num_global_atoms_K_prime = hdpObj.number_of_unique_global_atoms_used
    // num_tables_T_global = hdpObj.total_number_of_tables_across_all_groups (sum of num_clusters in each indDP that point to distinct global atoms)
    
    // The R code for UpdateGamma passes:
    // oldParam = dpObj$gamma (current global concentration)
    // n = sum(dpObj$numTables) (total #tables summed over all DPs, M_j in Teh et al notation)
    // nParams = dpObj$numGlobalParameters (number of unique dishes, K in Teh et al.)
    // priorParameters = dpObj$gammaPriors
    
    current_gamma = hdpObj.gamma
    total_tables_M = 0
    for each indDP in hdpObj.indDP_list:
        total_tables_M = total_tables_M + indDP.number_clusters_pointing_to_distinct_global_atoms 
        // This needs careful definition: sum over groups j of K_j (number of tables in group j)
        // Or, in update_g0, it's sum(globalParamTable$Freq) which is M (total tables)

    num_unique_global_atoms_K_global = hdpObj.number_of_active_global_atoms 
    gamma_prior_params = hdpObj.gamma_prior_parameters
    
    // Call the same underlying logic as UpdateAlpha.default
    // temp_dp_obj_for_gamma_update = create_dummy_dp_obj_with_params(current_gamma, total_tables_M, num_unique_global_atoms_K_global, gamma_prior_params)
    // hdpObj.gamma = UpdateAlpha.default(temp_dp_obj_for_gamma_update).alpha
    // Or more directly:
    x_gamma = sample_from_Beta(current_gamma + 1, total_tables_M)
    log_x_gamma = log(x_gamma)
    
    prior_shape_a_gamma = gamma_prior_params[0]
    prior_rate_b_gamma = gamma_prior_params[1]

    weight_numerator_gamma = prior_shape_a_gamma + num_unique_global_atoms_K_global - 1
    weight_denominator_term_gamma = total_tables_M * (prior_rate_b_gamma - log_x_gamma)
    
    pi1_gamma = 0
    if (weight_numerator_gamma + weight_denominator_term_gamma) > 0 and weight_numerator_gamma > 0:
        pi1_gamma = weight_numerator_gamma / (weight_numerator_gamma + weight_denominator_term_gamma)

    posterior_shape1_gamma = prior_shape_a_gamma + num_unique_global_atoms_K_global
    posterior_shape2_gamma = prior_shape_a_gamma + num_unique_global_atoms_K_global - 1
    posterior_rate_gamma = prior_rate_b_gamma - log_x_gamma

    if posterior_rate_gamma <= 0:
        new_gamma = current_gamma // Fallback
    else:
        random_uniform_gamma = sample_from_Uniform(0, 1)
        if random_uniform_gamma < pi1_gamma:
            if posterior_shape1_gamma > 0:
                new_gamma = sample_from_Gamma(shape = posterior_shape1_gamma, rate = posterior_rate_gamma)
            else: new_gamma = sample_from_Gamma(prior_shape_a_gamma, prior_rate_b_gamma)
        else:
            if posterior_shape2_gamma > 0:
                new_gamma = sample_from_Gamma(shape = posterior_shape2_gamma, rate = posterior_rate_gamma)
            else: 
                if num_unique_global_atoms_K_global == 0 and posterior_shape1_gamma > 0:
                     new_gamma = sample_from_Gamma(shape = posterior_shape1_gamma, rate = posterior_rate_gamma)
                else: new_gamma = sample_from_Gamma(prior_shape_a_gamma, prior_rate_b_gamma)
                
    hdpObj.gamma = new_gamma
    
    return hdpObj
