function MhParameterProposal(mixing_distribution_mdObj, current_params):
    // Proposes new parameters based on current_params and mdObj.mhStepSize
    // Specific to each kernel type.
    kernel_type = mdObj.type
    step_size = mdObj.mhStepSize // This might be a scalar or a structure

    if kernel_type == "beta":
        return MhParameterProposal.beta(current_params, step_size, mdObj.maxT)
    else if kernel_type == "weibull":
        return MhParameterProposal.weibull(current_params, step_size)
    // ... other non-conjugate kernels ...
    else:
        error "Unsupported kernel type for MhParameterProposal"

// Example: MhParameterProposal.beta
function MhParameterProposal.beta(current_beta_params, step_sizes, max_T):
    // current_beta_params = {mean_mu, precision_tau}
    // step_sizes might be {step_mu, step_tau}
    
    current_mu = current_beta_params.mean_mu
    current_tau = current_beta_params.precision_tau

    // Propose new mu: Normal random walk, then clip to [0, max_T]
    proposed_mu = current_mu + sample_from_Normal(mean=0, std_dev=step_sizes.mu)
    proposed_mu = max(0, min(proposed_mu, max_T))

    // Propose new tau: Normal random walk on log-scale or reflect to ensure positivity
    // R code adds Normal noise and takes abs value.
    proposed_tau_raw = current_tau + sample_from_Normal(mean=0, std_dev=step_sizes.tau)
    proposed_tau = abs(proposed_tau_raw) 
    if proposed_tau == 0: proposed_tau = very_small_positive_number // Avoid tau=0

    return {mean_mu = proposed_mu, precision_tau = proposed_tau}
