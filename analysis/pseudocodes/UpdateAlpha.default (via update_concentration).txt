function UpdateAlpha.default(dpObj):
    current_alpha = dpObj.alpha
    n = dpObj.number_of_data_points
    num_clusters_K = dpObj.number_clusters
    alpha_prior_params = dpObj.alpha_prior_parameters // e.g., [shape_a, rate_b] for Gamma prior

    // Escobar & West (1995) method
    // Draw auxiliary variable x
    x = sample_from_Beta(current_alpha + 1, n)

    // Calculate mixture weights for two Gamma distributions
    prior_shape_a = alpha_prior_params[0]
    prior_rate_b = alpha_prior_params[1]

    // Ensure log(x) is valid; if x is 0, this term might dominate or cause issues.
    // R code uses a term that implies log(x) is negative, so (rate_b - log(x)) > rate_b
    // Handle x=0 or x=1 carefully if they can occur.
    if x == 0: // or x is very close to 0
        // In this case, rate_b - log(x) becomes very large, pi1 approaches 1.
        // This leads to choosing the Gamma with shape prior_shape_a + num_clusters_K.
        pi1_numerator = prior_shape_a + num_clusters_K -1 // original has -1, but can be K or K-1
        // if pi1_denominator_term is very large, pi1 becomes small.
        // The R code is: (a + k -1) / ( (a+k-1) + n*(b-log(x)))
        // if x is small, log(x) is large negative, so (b-log(x)) is large positive. Denom is large. pi1 small.
        // Let's follow R: (a+K-1) / ( (a+K-1) + n*(b-log(x)) )
        // if x -> 0, log(x) -> -Inf, denom -> Inf, pi1 -> 0. Choose shape a+K-1
        // if x -> 1, log(x) -> 0, pi1 -> (a+K-1)/((a+K-1)+n*b).
        // The exact formulation might depend on which version of Escobar&West or Neal is used.
        // Let's assume x is in (0,1).
        
    log_x = log(x)
    
    // Weight for choosing the first Gamma component (shape_a + K) vs second (shape_a + K -1)
    // Based on Neal (2000) interpretation of Escobar & West (1995) (eq 10) for updating alpha
    // eta is the Beta(alpha+1, n) variable, which is `x` here.
    // alpha ~ pi_eta * G(a_k+k, b_k-log(eta)) + (1-pi_eta) * G(a_k+k-1, b_k-log(eta))
    // where pi_eta / (1-pi_eta) = (a_k+k-1) / (N * (b_k-log(eta)))
    // So, pi_eta = (a_k+k-1) / ( (a_k+k-1) + N*(b_k-log(eta)) )
    
    weight_numerator = prior_shape_a + num_clusters_K - 1
    weight_denominator_term = n * (prior_rate_b - log_x)
    
    // Handle cases where weight_numerator might be <= 0 if num_clusters_K=0 and prior_shape_a=1
    // or if num_clusters_K=0 and prior_shape_a < 1
    if weight_numerator <= 0: // Avoid division by zero or negative weights if components are non-positive
        // If K=0, it implies alpha should be small.
        // The second component (shape a+K-1) would be preferred.
        // This case may need specific handling based on prior choices for 'a'
        // For simplicity, if num_clusters_K=0, this implies a preference for a smaller alpha.
        // If prior_shape_a + num_clusters_K - 1 is problematic (e.g. <=0),
        // the sampler might default to one component or require prior_shape_a > 1 if K can be 0 or 1.
        // The R code seems robust to K=0 if a > 1.
        // If prior_shape_a + num_clusters_K - 1 <= 0, then pi1 could be 0 or negative.
        // If weight_numerator is 0 and weight_denominator_term > 0, pi1 = 0.
        // If prior_shape_a=1, num_clusters_K=0, then weight_numerator=0, pi1=0. New alpha from G(a+K-1, new_rate)
        // If prior_shape_a=1, num_clusters_K=1, then weight_numerator=1.
        pi1 = 0
        if (weight_numerator + weight_denominator_term) > 0 and weight_numerator > 0 : # Ensure denom is positive
             pi1 = weight_numerator / (weight_numerator + weight_denominator_term)
        // else if weight_numerator is 0, pi1 is 0. This leads to choosing Gamma with shape a+K-1 (or a+K if K=0 and shape is a)
        // The specific shape choice depends on how K=0 interacts with K-1.
        // R code: `shape1 = a + k`, `shape2 = a + k - 1`. If runif > pi_eta, use shape2.
        // If k=0, shape1 = a, shape2 = a-1. If pi_eta=0, use shape2.

    // Posterior Gamma parameters
    posterior_shape1 = prior_shape_a + num_clusters_K
    posterior_shape2 = prior_shape_a + num_clusters_K - 1
    posterior_rate = prior_rate_b - log_x
    
    if posterior_rate <= 0: // Rate must be positive
        // This can happen if prior_rate_b is small and log_x is large (x is small).
        // Fallback or re-evaluation needed. For now, assume valid posterior_rate.
        // Or, it implies data is inconsistent with prior, alpha should be very small.
        // Potentially draw from prior if posterior is invalid.
        // R code does not explicitly check for posterior_rate <= 0. rgamma would error.
        // This implies prior_rate_b should be large enough relative to typical -log(x) values.
        // For now, assume valid.
        new_alpha = current_alpha // Fallback, or draw from prior
    else:
        random_uniform = sample_from_Uniform(0, 1)
        if random_uniform < pi1: // With probability pi1
            if posterior_shape1 > 0:
                 new_alpha = sample_from_Gamma(shape = posterior_shape1, rate = posterior_rate)
            else: // If shape is not positive, Gamma is not well-defined. Fallback.
                 new_alpha = sample_from_Gamma(shape = prior_shape_a, rate = prior_rate_b) // Fallback to prior mean or sample
        else: // With probability 1-pi1
            if posterior_shape2 > 0:
                new_alpha = sample_from_Gamma(shape = posterior_shape2, rate = posterior_rate)
            else: // If shape is not positive (e.g. K=0, prior_shape_a=1 => posterior_shape2=0)
                 // If K=0 and prior_shape_a=1, posterior_shape2 = 0.
                 // In this situation, perhaps the component with K (i.e. posterior_shape1) is intended.
                 // The R code for update_concentration does:
                 // post_params1_shape_candidate1 <- priorParameters[1] + nParams
                 // if (runif(1) > pi1) { post_params1_shape_candidate1 <- priorParameters[1] + nParams - 1 }
                 // if (post_params1_shape_candidate1 <=0) # Draw from prior
                 // So it uses one shape variable and adjusts it.
                 // Let's stick to two distinct shapes and handle non-positivity.
                 new_alpha = sample_from_Gamma(shape = prior_shape_a, rate = prior_rate_b) // Fallback
                 // Or, if K=0, the term K-1 is problematic. The mixture should adapt.
                 // E.g. if K=0, only sample from G(a, b-log(x))
                 if num_clusters_K == 0 and posterior_shape1 > 0: // If K=0, effectively only one component exists
                      new_alpha = sample_from_Gamma(shape = posterior_shape1, rate = posterior_rate)


    dpObj.alpha = new_alpha
    return dpObj
