function PriorDraw(mixing_distribution_object_mdObj, num_samples_n):
    // This is a generic function that dispatches.
    kernel_type = mdObj.type
    prior_hyperparams = mdObj.prior_parameters

    if kernel_type == "normal": // (Normal-Inverse Gamma prior)
        return PriorDraw.normal(prior_hyperparams, num_samples_n)
    else if kernel_type == "beta": // (e.g., Uniform for mean, Inv-Gamma for scale)
        return PriorDraw.beta(mdObj, prior_hyperparams, num_samples_n) // mdObj for maxT
    else if kernel_type == "weibull":
        return PriorDraw.weibull(prior_hyperparams, num_samples_n)
    else if kernel_type == "mvnormal": // (e.g. Normal-Wishart)
        return PriorDraw.mvnormal(prior_hyperparams, num_samples_n, mdObj.dimension) // dim needed for Wishart
    else if kernel_type == "hierarchical":
         return PriorDraw.hierarchical(mdObj, num_samples_n) // mdObj has global atoms and weights
    // ... other kernels ...
    else:
        error "Unsupported kernel type for PriorDraw"

// Example: PriorDraw.normal (Normal-Inverse Gamma)
function PriorDraw.normal(prior_hyperparams, n_samples):
    // prior_hyperparams = {mu0, kappa0, alpha0, beta0}
    mu0 = prior_hyperparams.mu0
    kappa0 = prior_hyperparams.kappa0
    alpha0 = prior_hyperparams.alpha0 // shape for Inv-Gamma on variance (or Gamma on precision)
    beta0 = prior_hyperparams.beta0   // scale for Inv-Gamma on variance (or rate for Gamma on precision)

    sampled_means = empty_array(length = n_samples)
    sampled_std_devs = empty_array(length = n_samples)

    for i = 1 to n_samples:
        // 1. Draw precision lambda_val ~ Gamma(alpha0, beta0)
        precision_lambda_val = sample_from_Gamma(shape = alpha0, rate = beta0)
        variance_sigma_sq = 1 / precision_lambda_val
        sampled_std_devs[i] = sqrt(variance_sigma_sq)
        
        // 2. Draw mean mu ~ Normal(mu0, sqrt(variance_sigma_sq / kappa0))
        mean_std_dev = sqrt(variance_sigma_sq / kappa0)
        sampled_means[i] = sample_from_Normal(mean = mu0, std_dev = mean_std_dev)
        
    return list_of_parameter_arrays(means = sampled_means, std_devs = sampled_std_devs)
    // R returns: list(array(mus, dim=c(1,1,n)), array(sigmas, dim=c(1,1,n)))

// Example: PriorDraw.hierarchical
function PriorDraw.hierarchical(mdObj_hierarchical, n_samples):
    // mdObj_hierarchical contains the global atoms (theta_k_global) and global stick-breaking weights (pi_k_global)
    global_atoms = mdObj_hierarchical.theta_k_global 
    global_weights = mdObj_hierarchical.pi_k_global

    chosen_atom_indices = sample_from_categorical_weighted(indices = 1 to length(global_atoms), 
                                                            weights = global_weights, 
                                                            num_samples = n_samples, 
                                                            with_replacement = true)
    
    // Construct the list of parameter arrays from the chosen global atoms
    // E.g., if global_atoms is a list of {mean, variance} structures
    sampled_params_list = initialize_empty_parameter_list_structure_matching_atom_structure()

    for each sampled_atom_index in chosen_atom_indices:
        selected_atom_params = global_atoms[sampled_atom_index]
        // Append each component of selected_atom_params to the corresponding array in sampled_params_list
        // e.g., sampled_params_list.means.append(selected_atom_params.mean)
        //       sampled_params_list.variances.append(selected_atom_params.variance)
    
    // Ensure output is formatted as list of arrays, e.g. { means: [m1,m2..], variances: [v1,v2..] }
    // with each array element being a sample.
    // R: lapply(mdObj$theta_k, function(x) x[,, ind, drop = FALSE])
    // This implies theta_k is already a list of arrays, and we are selecting slices.
    
    // Pseudocode assuming theta_k is list of param arrays [param_type][dim1, dim2, atom_index]
    output_param_list = empty_list_of_arrays()
    for each param_array_type in mdObj_hierarchical.theta_k_global: // e.g., for "means", "variances"
        new_param_array_for_type = select_slices_from_array(param_array_type, indices = chosen_atom_indices, dimension_to_slice = 3)
        output_param_list.add(new_param_array_for_type)
        
    return output_param_list
