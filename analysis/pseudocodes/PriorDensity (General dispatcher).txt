function PriorDensity(mixing_distribution_mdObj, params_theta):
    // Calculates the prior probability density p(theta)
    // Specific to each kernel type.
    kernel_type = mdObj.type
    prior_hyperparams = mdObj.prior_parameters

    if kernel_type == "beta":
        return PriorDensity.beta(params_theta, prior_hyperparams, mdObj.maxT)
    else if kernel_type == "weibull":
        return PriorDensity.weibull(params_theta, prior_hyperparams)
    // ... other non-conjugate kernels ...
    else:
        error "Unsupported kernel type for PriorDensity"

// Example: PriorDensity.beta
function PriorDensity.beta(params_theta, prior_hyperparams_beta, max_T):
    // params_theta = {mean_mu, precision_tau}
    // prior_hyperparams_beta for beta = {mu_prior (e.g. Uniform on [0,maxT]), 
    //                                   tau_prior (e.g. InvGamma with shape, rate)}
    mu = params_theta.mean_mu
    tau = params_theta.precision_tau

    // Prior for mu: Uniform(0, max_T)
    // This implies prior_hyperparams_beta.mu_prior might be just the max_T for uniform.
    // Or it could specify params for a different distribution on mu.
    // Assuming Uniform(0, max_T) for mu based on R beta_uniform_gamma.R
    if mu < 0 or mu > max_T:
        density_mu = 0
    else:
        density_mu = 1 / max_T 
        
    // Prior for tau: e.g., Inverse-Gamma(shape_tau, rate_tau) which means 1/tau ~ Gamma(shape,rate)
    // Or Gamma(shape_tau, rate_tau) directly on tau. R code suggests tau ~ 1/Gamma.
    // So, p(tau) = p_gamma(1/tau) * (1/tau^2)
    // Or, if prior is dgamma(1/nu, shape=alpha0, rate=beta0) * (1/nu^2) for nu, then for nu=tau:
    // priorParameters for beta_uniform_gamma are alpha0, beta0 for InvGamma on tau (nu).
    // R: dgamma(1.0 / theta[[2]][,,i], priorParameters[1], priorParameters[2]) * (1.0/theta[[2]][,,i]^2)
    shape_tau_prior = prior_hyperparams_beta.shape_for_inv_gamma_tau // e.g., alpha0
    rate_tau_prior = prior_hyperparams_beta.rate_for_inv_gamma_tau   // e.g., beta0
    
    if tau <= 0:
        density_tau = 0
    else:
        density_tau = probability_density_Gamma(1/tau, shape=shape_tau_prior, rate=rate_tau_prior) * (1 / (tau^2))
        
    return density_mu * density_tau
