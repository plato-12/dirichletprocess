function ClusterComponentUpdate.nonconjugate(dpObj):
    n = dpObj.number_of_data_points
    alpha = dpObj.concentration_parameter
    m = dpObj.auxiliary_parameters_count
    data = dpObj.data
    mixing_distribution = dpObj.mixing_distribution
    cluster_labels = dpObj.cluster_labels
    points_per_cluster = dpObj.points_per_cluster
    cluster_parameters = dpObj.cluster_parameters
    number_clusters = dpObj.number_clusters

    for i = 1 to n:
        current_data_point = data[i]
        original_label = cluster_labels[i]

        // Temporarily remove point i from its cluster
        points_per_cluster[original_label] = points_per_cluster[original_label] - 1

        // Calculate probabilities for existing clusters
        probabilities = empty_list
        for k = 1 to number_clusters:
            if points_per_cluster[k] > 0: // Only consider non-empty clusters
                likelihood_val = Likelihood(mixing_distribution, current_data_point, cluster_parameters[k])
                probabilities.add(points_per_cluster[k] * likelihood_val)
            else:
                // If this was the original_label and it's now empty, its prob contribution is handled by aux params
                // or effectively zero if it's some other pre-existing empty cluster (should not happen ideally)
                probabilities.add(0) 

        // Handle auxiliary parameters
        aux_params = empty_list
        if points_per_cluster[original_label] == 0:
            // Original cluster became empty, reuse its parameters as one of the m auxiliary ones
            aux_params.add(cluster_parameters[original_label])
            if m > 1:
                aux_params.append_all(PriorDraw(mixing_distribution, m - 1))
        else:
            // Original cluster still has points, draw m new auxiliary parameters
            aux_params.append_all(PriorDraw(mixing_distribution, m))
        
        // Calculate probabilities for new clusters via auxiliary parameters
        for j = 1 to length(aux_params):
            likelihood_val = Likelihood(mixing_distribution, current_data_point, aux_params[j])
            probabilities.add((alpha / m) * likelihood_val)

        // Normalize probabilities
        normalized_probabilities = probabilities / sum(probabilities)
        // Handle case where all probabilities are zero (e.g., assign to a new cluster or handle error)
        if sum(probabilities) == 0:
            // Default to creating a new cluster using the first auxiliary parameter
            // or a more sophisticated rule might be needed.
            // For simplicity, assume one aux param always has non-zero likelihood or is chosen.
            // The R code assigns a label and ClusterLabelChange handles param init.
            // Here, we'll assume a valid new_label is chosen.
            chosen_index = index_of_first_aux_param_in_probabilities_array 
        else:
            chosen_index = sample_from_categorical(normalized_probabilities)
        
        // Determine new_label and if it's from aux_params
        is_new_from_aux = false
        if chosen_index > number_clusters: // Index corresponds to one of the aux_params
            new_label_for_logic = number_clusters + (chosen_index - number_clusters) // This maps to aux_params index
            is_new_from_aux = true
            selected_aux_param_for_new_cluster = aux_params[chosen_index - number_clusters]
        else: // Index corresponds to an existing cluster
            new_label_for_logic = chosen_index
            selected_aux_param_for_new_cluster = null 
            // (though R's ClusterLabelChange might draw if newLabel is an empty existing slot)

        // Call ClusterLabelChange to update dpObj state
        // ClusterLabelChange will handle:
        // - Setting dpObj.cluster_labels[i]
        // - Updating dpObj.points_per_cluster for old and new labels
        // - Adjusting dpObj.number_clusters
        // - Modifying dpObj.cluster_parameters (removing empty, adding new with selected_aux_param_for_new_cluster if needed)
        dpObj = ClusterLabelChange(dpObj, 
                                   data_point_index = i, 
                                   new_label_raw_index = chosen_index, // The sampled index from combined prob list
                                   current_label_original = original_label,
                                   auxiliary_parameters_if_chosen = selected_aux_param_for_new_cluster,
                                   is_new_from_aux_flag = is_new_from_aux) // Flag to guide ClusterLabelChange

    return dpObj
