function Predictive(mixing_distribution_object_mdObj, data_point_x):
    // Calculates p(x | G_0) = integral k(x | theta) dG_0(theta)
    // Typically only for conjugate models.
    kernel_type = mdObj.type

    if mdObj.is_conjugate:
        if kernel_type == "normal":
            return Predictive.normal(mdObj, data_point_x)
        else if kernel_type == "exponential":
            return Predictive.exponential(mdObj, data_point_x)
        else if kernel_type == "mvnormal":
            return Predictive.mvnormal(mdObj, data_point_x)
        // ... other conjugate kernels ...
        else:
            error "Unsupported conjugate kernel for Predictive"
    else:
        // For non-conjugate, this is not typically used directly in Neal's Alg 8 for new clusters.
        // It might return an error or a numerical approximation if available.
        error "Predictive usually not analytically available for non-conjugate models"

// Example: Predictive.normal (Normal data, Normal-Inverse Gamma prior)
function Predictive.normal(mdObj, x_point):
    // mdObj.priorParameters = {mu0, kappa0, alpha0, beta0}
    mu0 = mdObj.prior_parameters.mu0
    kappa0 = mdObj.prior_parameters.kappa0
    alpha0 = mdObj.prior_parameters.alpha0
    beta0 = mdObj.prior_parameters.beta0

    // This results in a non-standardized Student's t-distribution for x_point
    // The R code calculates posterior parameters as if x_point was observed:
    // PosteriorParameters(mdObj, x_point) would yield mu_n, kappa_n, alpha_n, beta_n
    // Let's assume x_point is a single scalar for simplicity
    // These are posterior predictive parameters for a single point x_point
    kappa_n_for_x = kappa0 + 1
    mu_n_for_x = (kappa0 * mu0 + x_point) / kappa_n_for_x
    alpha_n_for_x = alpha0 + 0.5
    beta_n_for_x = beta0 + (kappa0 * (x_point - mu0)^2) / (2 * kappa_n_for_x)
    
    // Predictive likelihood for x_point (density of a non-standard Student's t)
    // Formula: Gamma(alpha_n) / Gamma(alpha0) * //          (beta0^alpha0) / (beta_n^alpha_n) * //          sqrt(kappa0 / (kappa_n * 2 * PI)) -- (the 2PI factor for student-t density)
    // R code is: (gamma(alpha_n)/gamma(alpha0)) * ((beta0^alpha0)/(beta_n^alpha_n)) * sqrt(kappa0/kappa_n)
    // This matches the structure if we omit the 1/sqrt(2PI) constant if it cancels or is absorbed elsewhere.
    // Let's follow R's structure:
    
    term1 = Gamma_function(alpha_n_for_x) / Gamma_function(alpha0)
    term2 = (beta0^alpha0) / (beta_n_for_x^alpha_n_for_x) // Make sure powers are correct
    term3 = sqrt(kappa0 / kappa_n_for_x)

    return term1 * term2 * term3
